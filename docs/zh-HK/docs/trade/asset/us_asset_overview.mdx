package main

import (
	"context"
	"fmt"
	"log"

	"github.com/longbridge/openapi-go/config"
	"github.com/longbridge/openapi-go/oauth"
	"github.com/longbridge/openapi-go/trade"
)

func main() {
	o := oauth.New("your-client-id").
		OnOpenURL(func(url string) { fmt.Println("Open this URL to authorize:", url) })
	if err := o.Build(context.Background()); err != nil {
		log.Fatal(err)
	}
	conf, err := config.New(config.WithOAuthClient(o))
	if err != nil {
		log.Fatal(err)
	}
	c, err := trade.NewFromCfg(conf)
	if err != nil {
		log.Fatal(err)
	}
	defer c.Close()
	resp, err := c.USAssetOverview(context.Background())
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("%+v\n", resp)
}

## Schemas

### USAssetOverview

<a id="USAssetOverview"></a>

| 名稱 | 類型 | 必填 | 描述 |
| ---- | ---- | ---- | ---- |
| account_type | string | 是 | 賬戶類型標識，例如 `"1"` |
| asset_timestamp | string | 是 | 資產數據快照時間，以數字字符串表示的 Unix 秒（例如 `"1792661597"`）；SDK 返回整數 |
| cash_buy_power | string | 是 | 可用買入力（现金） |
| overnight_buy_power | string | 是 | 隔夜買入力 |
| currency | string | 是 | 基礎貨幣 |
| cash_list | USCashEntry[] | 否 | 按貨幣分列的現金餘額 |
| stock_list | USStockEntry[] | 否 | 股票持倉 |
| option_list | object[] | 否 | 期權持倉 |
| crypto_list | USCryptoEntry[] | 否 | 加密貨幣持倉 |
| multi_leg | object | 否 | 多腿（期權組合）持倉，位於 `holdings` 列表中。由 HTTP 接口返回 |

### USCashEntry

<a id="USCashEntry"></a>

| 名稱 | 類型 | 描述 |
| ---- | ---- | ---- |
| currency | string | 貨幣代碼 |
| total_cash | string | 總現金 |
| settled_cash | string | 已結算現金 |
| total_amount | string | 含未結算的總金額 |
| outstanding | string | 未結算金額 |
| frozen_buy_cash | string | 待成交買單凍結金額 |

### USStockEntry

<a id="USStockEntry"></a>

| 名稱 | 類型 | 描述 |
| ---- | ---- | ---- |
| symbol | string | 不帶市場後綴的股票代碼，例如 `MULL` |
| full_symbol | string | 完整代碼，例如 `MULL.US` |
| counter_id | string | 內部標的標識，例如 `ETF/US/MULL`。由 HTTP 接口返回 |
| asset_type | string | 資產類型，例如 `STOCK` |
| quantity | string | 持有數量 |
| currency | string | 貨幣代碼 |
| average_cost | string | 平均持倉成本價 |
| market | string | 市場標識 |
| trade_status | string | 交易狀態代碼，例如 `201` |
| prev_close | string | 上一收盤價 |
| last_done | string | 最新成交價 |
| market_price | string | 當前市場價格 |
| today_pl | string | 當日盈虧 |
| name | string | 證券名稱 |
| position_side | string | 持倉方向（多/空） |
| industry_name | string | 行業/板塊名稱 |

### USCryptoEntry

<a id="USCryptoEntry"></a>

| 名稱 | 類型 | 描述 |
| ---- | ---- | ---- |
| symbol | string | 不帶市場後綴的加密貨幣交易對代碼，例如 `BTCUSD`；調用其他接口時請使用 `BTCUSD.BKKT` |
| counter_id | string | 內部標的標識，例如 `VA/BKKT/BTCUSD`。由 HTTP 接口返回 |
| market | string | 市場：`BKKT`。由 HTTP 接口返回 |
| name | string | 加密貨幣名稱，例如 `Bitcoin`。由 HTTP 接口返回 |
| quantity | string | 持有數量。由 HTTP 接口返回 |
| average_cost | string | 平均持倉成本價 |
| currency | string | 計價貨幣 |
| asset_type | string | 資產類型：`CRYPTO` |
| industry_name | string | 行業/分類名稱 |

HTTP 接口返回的加密貨幣記錄除 `full_symbol` 外與股票記錄字段相同；SDK 的加密貨幣類型可能只包含上表中未標註「由 HTTP 接口返回」的字段。

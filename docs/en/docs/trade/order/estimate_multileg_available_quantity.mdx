This API is used to estimate the maximum purchase quantity and the margin impact of a multi-leg option combination strategy before submitting the order.

Constraints:

- Only the US market is supported.
- All legs must share the same underlying.
- Each leg must be an option or the underlying stock itself.
- Duplicate legs (the same symbol with the same side) are not allowed.
- All legs must share the same settlement currency.

<CliCommand>
longbridge estimate-multileg --side buy --strategy VerticalCallSpread --quantity 1 --leg QQQ260731C764000.US --leg QQQ260731C767000.US --price 1.5
</CliCommand>

### Parameters

> Content-Type: application/json; charset=utf-8

| Name               | Type     | Required | Description                                                                                                                                                        |
| ------------------ | -------- | -------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| side               | string   | YES      | Order side of the combination<br/><br/> **Enum Value:**<br/> `Buy`<br/> `Sell`                                                                                     |
| order_type         | string   | YES      | [Order Type](../trade-definition#ordertype)                                                                                                                        |
| submitted_price    | string   | NO       | Estimated order price, example: `1.5`<br/><br/> Required for limit order types such as `LO`                                                                        |
| submitted_quantity | string   | YES      | Order quantity (number of combinations), example: `1`                                                                                                              |
| strategy           | string   | YES      | Multi-leg strategy<br/><br/> **Enum Value:**<br/> `CoveredCall` - Covered stock<br/> `CoveredPut` - Covered stock<br/> `VerticalCallSpread` - Vertical spread<br/> `VerticalPutSpread` - Vertical spread<br/> `Collar` - Collar<br/> `Straddle` - Straddle<br/> `Strangle` - Strangle<br/> `CalendarCallSpread` - Calendar<br/> `CalendarPutSpread` - Calendar |
| legs               | object[] | YES      | Legs of the combination                                                                                                                                            |
| ∟ symbol           | string   | YES      | Option or underlying stock symbol, use `ticker.region` format, example: `QQQ260731C764000.US`                                                                      |
| ∟ side             | string   | NO       | Leg order side.<br/><br/>Reserved field, not in use for now.<br/><br/> **Enum Value:**<br/> `Buy`<br/> `Sell` |

package main

import (
	"context"
	"fmt"
	"log"

	"github.com/longbridge/openapi-go/config"
	"github.com/longbridge/openapi-go/oauth"
	"github.com/longbridge/openapi-go/trade"
	"github.com/shopspring/decimal"
)

func main() {
	o := oauth.New("your-client-id").
		OnOpenURL(func(url string) { fmt.Println("Open this URL to authorize:", url) })
	if err := o.Build(context.Background()); err != nil {
		log.Fatal(err)
	}
	conf, err := config.New(config.WithOAuthClient(o))
	if err != nil {
		log.Fatal(err)
	}
	tctx, err := trade.NewFromCfg(conf)
	if err != nil {
		log.Fatal(err)
	}
	defer tctx.Close()
	resp, err := tctx.EstimateMultiLegAvailableQuantity(context.Background(), &trade.GetEstimateMultiLegAvailableQuantity{
		Side:              trade.OrderSideBuy,
		OrderType:         trade.OrderTypeLO,
		SubmittedQuantity: decimal.RequireFromString("1"),
		Strategy:          trade.MultiLegStrategyVerticalCallSpread,
		SubmittedPrice:    decimal.RequireFromString("1.5"),
		Legs: []trade.EstimateMultiLegOrderLeg{
			{Symbol: "QQQ260731C764000.US"},
			{Symbol: "QQQ260731C767000.US"},
		},
	})
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(resp)
}

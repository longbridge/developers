package main

import (
	"context"
	"fmt"
	"log"

	"github.com/longbridge/openapi-go/config"
	"github.com/longbridge/openapi-go/oauth"
	"github.com/longbridge/openapi-go/trade"
)

func main() {
	o := oauth.New("your-client-id").
		OnOpenURL(func(url string) { fmt.Println("Open this URL to authorize:", url) })
	if err := o.Build(context.Background()); err != nil {
		log.Fatal(err)
	}
	conf, err := config.New(config.WithOAuthClient(o))
	if err != nil {
		log.Fatal(err)
	}
	c, err := trade.NewFromCfg(conf)
	if err != nil {
		log.Fatal(err)
	}
	defer c.Close()
	resp, err := c.USAssetOverview(context.Background())
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("%+v\n", resp)
}

## Schemas

### USAssetOverview

<a id="USAssetOverview"></a>

| 名称 | 类型 | 必填 | 描述 |
| ---- | ---- | ---- | ---- |
| account_type | string | 是 | 账户类型标识，例如 `"1"` |
| asset_timestamp | string | 是 | 资产数据快照时间，以数字字符串表示的 Unix 秒（例如 `"1792661597"`）；SDK 返回整数 |
| cash_buy_power | string | 是 | 可用买入力（现金） |
| overnight_buy_power | string | 是 | 隔夜买入力 |
| currency | string | 是 | 基础货币 |
| cash_list | USCashEntry[] | 否 | 按货币分列的现金余额 |
| stock_list | USStockEntry[] | 否 | 股票持仓 |
| option_list | object[] | 否 | 期权持仓 |
| crypto_list | USCryptoEntry[] | 否 | 加密货币持仓 |
| multi_leg | object | 否 | 多腿（期权组合）持仓，位于 `holdings` 列表中。由 HTTP 接口返回 |

### USCashEntry

<a id="USCashEntry"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| currency | string | 货币代码 |
| total_cash | string | 总现金 |
| settled_cash | string | 已结算现金 |
| total_amount | string | 含未结算的总金额 |
| outstanding | string | 未结算金额 |
| frozen_buy_cash | string | 待成交买单冻结金额 |

### USStockEntry

<a id="USStockEntry"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| symbol | string | 不带市场后缀的股票代码，例如 `MULL` |
| full_symbol | string | 完整代码，例如 `MULL.US` |
| counter_id | string | 内部标的标识，例如 `ETF/US/MULL`。由 HTTP 接口返回 |
| asset_type | string | 资产类型，例如 `STOCK` |
| quantity | string | 持有数量 |
| currency | string | 货币代码 |
| average_cost | string | 平均持仓成本价 |
| market | string | 市场标识 |
| trade_status | string | 交易状态代码，例如 `201` |
| prev_close | string | 上一收盘价 |
| last_done | string | 最新成交价 |
| market_price | string | 当前市场价格 |
| today_pl | string | 当日盈亏 |
| name | string | 证券名称 |
| position_side | string | 持仓方向（多/空） |
| industry_name | string | 行业/板块名称 |

### USCryptoEntry

<a id="USCryptoEntry"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| symbol | string | 不带市场后缀的加密货币交易对代码，例如 `BTCUSD`；市场在 `market` 字段中单独返回。其他接口（包括[美股交易流水](/zh-CN/docs/trade/asset/us_transaction_history)）使用 `TICKER.MARKET` 格式 `BTCUSD.BKKT` |
| counter_id | string | 内部标的标识，例如 `VA/BKKT/BTCUSD`。由 HTTP 接口返回 |
| market | string | 市场：`BKKT` |
| name | string | 加密货币名称，例如 `Bitcoin` |
| quantity | string | 持有数量 |
| average_cost | string | 平均持仓成本价 |
| currency | string | 计价货币 |
| asset_type | string | 资产类型：`CRYPTO` |
| industry_name | string | 行业/分类名称 |
| position_side | string | 持仓方向，例如 `LONG` |
| trade_status | string | 交易状态代码，例如 `105` |
| prev_close | string | 上一收盘价 |
| last_done | string | 最新成交价 |
| market_price | string | 当前市场价格 |
| today_pl | string | 当日盈亏 |

HTTP 接口返回的加密货币记录除 `full_symbol` 外与股票记录字段相同。SDK 的加密货币类型包含上表除 `counter_id` 外的全部字段。

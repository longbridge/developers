package main

import (
	"context"
	"fmt"
	"log"

	"github.com/longbridge/openapi-go/config"
	"github.com/longbridge/openapi-go/oauth"
	"github.com/longbridge/openapi-go/trade"
)

func main() {
	o := oauth.New("your-client-id").
		OnOpenURL(func(url string) { fmt.Println("Open this URL to authorize:", url) })
	if err := o.Build(context.Background()); err != nil {
		log.Fatal(err)
	}
	conf, err := config.New(config.WithOAuthClient(o))
	if err != nil {
		log.Fatal(err)
	}
	c, err := trade.NewFromCfg(conf)
	if err != nil {
		log.Fatal(err)
	}
	defer c.Close()
	resp, err := c.USCashAndBuyPower(context.Background(), "USD")
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("%+v\n", resp)
}

## Schemas

### USCashAndBuyPower

<a id="USCashAndBuyPower"></a>

| 名称 | 类型 | 必填 | 描述 |
| ---- | ---- | ---- | ---- |
| buy_limit_info | [USBuyLimitInfo](#USBuyLimitInfo) | 是 | 购买力 |
| cash_list | [USCashDetail](#USCashDetail)[] | 是 | 按币种分列的现金余额 |
| margin_status | [USMarginStatus](#USMarginStatus) | 是 | 保证金账户状态 |

### USBuyLimitInfo

<a id="USBuyLimitInfo"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| currency | string | 货币 |
| cash_buy_power | string | 现金购买力 |
| overnight_buy_power | string | 隔夜购买力 |
| day_trade_buy_power | string | 日内交易购买力 |
| instant_buy_power | string | 即时购买力 |
| intraday_margin_buy_power | string | 日内保证金购买力 |
| option_buy_power | string | 期权购买力 |
| crypto_buy_power | string | 加密货币购买力 |
| mme | string | 用于计算购买力的维持保证金盈余 |

### USCashDetail

<a id="USCashDetail"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| currency | string | 货币 |
| total_amount | string | 现金总额 |
| available_cash | string | 可用现金 |
| withdraw_cash | string | 可取现金 |
| frozen_cash | string | 待成交订单冻结的现金 |
| settled_cash | string | 已结算现金 |
| outstanding | string | 未结算金额 |
| external_amount | string | 新股认购或基金申购占用的金额 |

### USMarginStatus

<a id="USMarginStatus"></a>

| 名称 | 类型 | 描述 |
| ---- | ---- | ---- |
| currency | string | 货币 |
| nlv | string | 净清算价值 |
| margin_equity | string | 保证金权益 |
| margin_maintenance_excess | string | 维持保证金盈余 |
| mmr | string | 维持保证金要求 |
| reg_t_excess | string | Reg T 盈余 |
| sma | string | 特别备忘账户（SMA） |

该接口用于在提交多腿期权组合订单前，预估该组合策略的最大购买数量及保证金变化。

限制条件：

- 仅支持美股市场。
- 所有腿必须属于同一标的。
- 每条腿只能是期权或其正股。
- 不允许重复腿（symbol 与买卖方向均相同）。
- 所有腿的结算货币必须相同。

<CliCommand>
longbridge estimate-multileg --side buy --strategy VerticalCallSpread --quantity 1 --leg QQQ260731C764000.US --leg QQQ260731C767000.US --price 1.5
</CliCommand>

### Parameters

> Content-Type: application/json; charset=utf-8

| Name               | Type     | Required | Description                                                                                                                                                        |
| ------------------ | -------- | -------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| side               | string   | YES      | 组合买卖方向<br/><br/> **可选值：**<br/> `Buy` - 买入<br/> `Sell` - 卖出                                                                                           |
| order_type         | string   | YES      | [订单类型](../trade-definition#ordertype)                                                                                                                          |
| submitted_price    | string   | NO       | 预估下单价格，例如：`1.5`<br/><br/> `LO` 等限价类型订单必填                                                                                                         |
| submitted_quantity | string   | YES      | 下单数量（组合数量），例如：`1`                                                                                                                                     |
| strategy           | string   | YES      | 多腿策略<br/><br/> **可选值：**<br/> `CoveredCall` - 股票担保<br/> `CoveredPut` - 股票担保<br/> `VerticalCallSpread` - 垂直策略<br/> `VerticalPutSpread` - 垂直策略<br/> `Collar` - 领式策略<br/> `Straddle` - 跨式策略<br/> `Strangle` - 宽跨式策略<br/> `CalendarCallSpread` - 日历策略<br/> `CalendarPutSpread` - 日历策略 |
| legs               | object[] | YES      | 组合的各腿                                                                                                                                                          |
| ∟ symbol           | string   | YES      | 期权或正股 symbol，使用 `ticker.region` 格式，例如：`QQQ260731C764000.US`                                                                                          |
| ∟ side             | string   | NO       | 该腿的买卖方向。<br/><br/>预留字段，暂不使用。<br/><br/> **可选值：**<br/> `Buy` - 买入<br/> `Sell` - 卖出        |

package main

import (
	"context"
	"fmt"
	"log"

	"github.com/longbridge/openapi-go/config"
	"github.com/longbridge/openapi-go/oauth"
	"github.com/longbridge/openapi-go/trade"
	"github.com/shopspring/decimal"
)

func main() {
	o := oauth.New("your-client-id").
		OnOpenURL(func(url string) { fmt.Println("Open this URL to authorize:", url) })
	if err := o.Build(context.Background()); err != nil {
		log.Fatal(err)
	}
	conf, err := config.New(config.WithOAuthClient(o))
	if err != nil {
		log.Fatal(err)
	}
	tctx, err := trade.NewFromCfg(conf)
	if err != nil {
		log.Fatal(err)
	}
	defer tctx.Close()
	resp, err := tctx.EstimateMultiLegAvailableQuantity(context.Background(), &trade.GetEstimateMultiLegAvailableQuantity{
		Side:              trade.OrderSideBuy,
		OrderType:         trade.OrderTypeLO,
		SubmittedQuantity: decimal.RequireFromString("1"),
		Strategy:          trade.MultiLegStrategyVerticalCallSpread,
		SubmittedPrice:    decimal.RequireFromString("1.5"),
		Legs: []trade.EstimateMultiLegOrderLeg{
			{Symbol: "QQQ260731C764000.US"},
			{Symbol: "QQQ260731C767000.US"},
		},
	})
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(resp)
}
